Func Main()
    Local $n, $i, $j, $k, $x
    $n = 10
    While $n > 0
        $n = $n - 3
    WEnd
    Do
        $n = $n + 1
    Until $n >= 5
    For $i = 1 To 10
        ConsoleWrite($i & @CRLF)
    Next
    For $j = 0 To $n - 1
        ConsoleWrite($j & @CRLF)
    Next
    $k = 0
    While $k < $n
        ConsoleWrite($k & @CRLF)
        $k = $k + 2
    WEnd
    While True
        work()
        ExitLoop
    WEnd
    For $x In $items
        ConsoleWrite($x & @CRLF)
    Next
    While $n > 100
    WEnd
EndFunc
