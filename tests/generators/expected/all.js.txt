function main() {
    let n, s, count, j;
    n = Number(prompt("n"));
    s = 0;
    count = 0;
    for (let i = 1; i <= n; i++) {
        if (i < 5) {
            for (j = 0; j < i; j = j + 1) {
                switch (j) {
                    case 0: {
                        continue;
                    }
                    case 1:
                    case 2: {
                        while (s < 100) {
                            s = s + j;
                            if (s > 50) {
                                break;
                            }
                        }
                        break;
                    }
                    default: {
                        do {
                            s = s - 1;
                            if (s < 10) {
                                continue;
                            }
                            console.log(`s = ${s}`);
                        } while (s > 0);
                    }
                }
            }
        } else {
            for (const x of items) {
                show(x);
                if (x > i) {
                    return x;
                }
            }
        }
    }
    console.log(`s = ${s}, count = ${count}`);
}

main();
