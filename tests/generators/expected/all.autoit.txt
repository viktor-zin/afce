Func Main()
    Local $n, $s, $count, $i, $j, $x
    $n = Number(InputBox("Input", "n"))
    $s = 0
    $count = 0
    For $i = 1 To $n
        If $i < 5 Then
            $j = 0
            While $j < $i
                Switch $j
                    Case 0
                        $j = $j + 1
                        ContinueLoop
                    Case 1, 2
                        While $s < 100
                            $s = $s + $j
                            If $s > 50 Then
                                ExitLoop
                            EndIf
                        WEnd
                    Case Else
                        Do
                            $s = $s - 1
                            If $s < 10 Then
                                ContinueLoop
                            EndIf
                            ConsoleWrite("s = " & $s & @CRLF)
                        Until $s <= 0
                EndSwitch
                $j = $j + 1
            WEnd
        Else
            For $x In $items
                show($x)
                If $x > $i Then
                    Return $x
                EndIf
            Next
        EndIf
    Next
    ConsoleWrite("s = " & $s & ", count = " & $count & @CRLF)
EndFunc
