program Algorithm;

var
  a, i, x, j: Integer;

begin
  while a < b do
  begin
    if a > 100 then
    begin
      Break;
    end;
    if a < 0 then
    begin
      Continue;
    end;
    a := a + 1;
  end;
  repeat
    a := a + 1;
    if a > 5 then
    begin
      Continue;
    end;
    if a > 50 then
    begin
      Break;
    end;
  until a >= b;
  for i := 1 to 10 do
  begin
    if i > 5 then
    begin
      Break;
    end
    else
    begin
      Continue;
    end;
  end;
  i := 0;
  while i < 10 do
  begin
    if i > 8 then
    begin
      Break;
    end;
    if i > 5 then
    begin
      i := i + 1;
      Continue;
    end;
    WriteLn(i);
    i := i + 1;
  end;
  for x in items do
  begin
    if x < 0 then
    begin
      Continue;
    end
    else
    begin
      Break;
    end;
  end;
  j := 0;
  while j < 3 do
  begin
    case j of
      0: begin
        j := j + 1;
        Continue;
      end;
      1: begin
        WriteLn('one');
      end;
      else begin
        WriteLn(j);
      end;
    end;
    j := j + 1;
  end;
  if a > b then
  begin
    Halt(a);
  end
  else
  begin
    Exit;
  end;
end.
