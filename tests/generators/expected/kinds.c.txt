#include <stdbool.h>
#include <stdio.h>

double stats(const char *s, char c, double x, int n) {
    int k, sum, done, t;
    double avg, half;
    printf("%s|%c|%g|%d\n", s, c, x, n);
    scanf("%d", &k);
    sum = 0;
    while (k) {
        sum = sum + k % 10;
        k = k / 10;
    }
    done = 0;
    do {
        done = 1;
    } while (!done && n % 2 == 0);
    avg = (double) sum / n;
    half = x / 2;
    t = (int) (avg * 10) % 7 * 2;
    if (n) {
        printf("%g %g %d\n", avg, half, t);
    }
    switch (sum % 3) {
        case 0: {
            printf("divisible\n");
            break;
        }
    }
    return avg / 2;
}
