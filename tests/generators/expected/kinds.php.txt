<?php

function stats($s, $c, $x, $n)
{
    echo $s, "|", $c, "|", $x, "|", $n, PHP_EOL;
    $k = (int) trim(fgets(STDIN));
    $sum = 0;
    while ($k) {
        $sum = $sum + $k % 10;
        $k = intdiv($k, 10);
    }
    $done = 0;
    do {
        $done = 1;
    } while (!$done && $n % 2 == 0);
    $avg = (float) $sum / $n;
    $half = $x / 2;
    $t = (int) ($avg * 10) % 7 * 2;
    if ($n) {
        echo $avg, " ", $half, " ", $t, PHP_EOL;
    }
    switch ($sum % 3) {
        case 0:
            echo "divisible", PHP_EOL;
            break;
    }
    return $avg / 2;
}
