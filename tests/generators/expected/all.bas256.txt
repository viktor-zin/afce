input n
s = 0
count = 0
for i = 1 to n
    if i < 5 then
        j = 0
        while j < i
            begin case
                case j = 0
                    j = j + 1
                    continue while
                case j = 1 or j = 2
                    while s < 100
                        s = s + j
                        if s > 50 then
                            exit while
                        end if
                    end while
                else
                    do
                        s = s - 1
                        if s < 10 then
                            continue do
                        end if
                        print "s = "; s
                    until s <= 0
            end case
            j = j + 1
        end while
    else
        for x_i = 0 to items[?] - 1
            x = items[x_i]
            call show(x)
            if x > i then
                end
            end if
        next x_i
    end if
next i
print "s = "; s; ", count = "; count
end
