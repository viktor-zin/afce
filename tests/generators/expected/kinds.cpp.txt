#include <iostream>

double stats(const char *s, char c, double x, int n) {
    int k, sum, done, t;
    double avg, half;
    std::cout << s << "|" << c << "|" << x << "|" << n << std::endl;
    std::cin >> k;
    sum = 0;
    while (k) {
        sum = sum + k % 10;
        k = k / 10;
    }
    done = 0;
    do {
        done = 1;
    } while (!done && n % 2 == 0);
    avg = (double) sum / n;
    half = x / 2;
    t = (int) (avg * 10) % 7 * 2;
    if (n) {
        std::cout << avg << " " << half << " " << t << std::endl;
    }
    switch (sum % 3) {
        case 0: {
            std::cout << "divisible" << std::endl;
            break;
        }
    }
    return avg / 2;
}
