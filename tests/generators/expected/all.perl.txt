use warnings;
use utf8;

    for my $i (1..$n) {
        if ($i < 5) {
            for ($j = 0; $j < $i; $j = $j + 1) {
                if ($j == 0) {
                    next;
                } elsif ($j == 1 || $j == 2) {
                    while ($s < 100) {
                        $s = $s + $j;
                        if ($s > 50) {
                            last;
                        }
                    }
                } else {
                    while (1) {
                        $s = $s - 1;
                        if ($s < 10) {
                            last unless $s > 0;
                            next;
                        }
                        print("s = ", $s, "\n");
